⚠️ **`--index-url` 必须写。** 全局配的清华 PyPI 镜像里那个 `torch` 是通用打包（带全套 `nvidia-*` 依赖），在这台机器上纯属浪费两三个 G；官方 `whl/cu128` 才是 Windows 自包含轮子。

验证与实测：

```text

计算能力 (8, 6)                       # Ampere sm_86
显存 mem_get_info() → 空闲 5.01 GB / 总 6.0 GB
```

性能对比（20 次 1014×1014 矩阵乘，正是论文 1,014 栋建筑的量级）：

```text
CPU   0.1323 s
CUDA  0.0070 s      → 约 19 倍
200 轮 logsumexp(1014,1014)  0.141 s
```

**这个结果修正了原先的判断**：本来以为"论文模型极小、GPU 帮不上忙"，实测稠密矩阵这块 GPU 快一个数量级。因此 G08（谱半径与发散实验）和 G16（四网络训练对比）可以把规模放大到 2000～3000 节点，作为复现报告里额外的一节；`torch_geometric` 也值得走 CUDA 版而不是急着用 dense 降级方案。

`D:\env_check.py` —— 用"本月真要做的计算"代替 `import` 冒烟：

```python
"""geoirl 环境冒烟测试：不测 import，测本月真正要做的计算。
跑法：conda activate geoirl && python D:\env_check.py
"""
import numpy as np
import shapely
import pyogrio
import geopandas as gpd
from shapely.geometry import Polygon
import networkx as nx
from lifelines import KaplanMeierFitter

print("=" * 60)
print("版本")
print("=" * 60)
print(f"numpy       {np.__version__}")
print(f"shapely     {shapely.__version__}  GEOS {shapely.geos_version}")
print(f"geopandas   {gpd.__version__}")
print(f"pyogrio     {pyogrio.__version__}  GDAL {pyogrio.__gdal_version__}")
print(f"networkx    {nx.__version__}")

# ---------- 1. 三栋建筑：面积与质心（G15 特征矩阵的原料） ----------
polys = [
    Polygon([(0, 0), (10, 0), (10, 10), (0, 10)]),      # 100
    Polygon([(20, 0), (40, 0), (40, 15), (20, 15)]),    # 300
    Polygon([(50, 50), (58, 50), (58, 60), (50, 60)]),  # 80
]
gdf = gpd.GeoDataFrame({"bid": [1, 2, 3]}, geometry=polys, crs="EPSG:2039")
# 教训：列名一律用中括号取，gdf.cx 会命中几何坐标访问器
gdf["area"] = gdf.geometry.area
gdf["cx"] = gdf.geometry.centroid.x
gdf["cy"] = gdf.geometry.centroid.y
areas = gdf["area"].to_numpy()
assert np.allclose(areas, [100.0, 300.0, 80.0]), areas
print(f"\n[1] 建筑面积 {areas.tolist()}  ✅")

# ---------- 2. 质心距离矩阵（SGS 的输入，论文式(1) 的 D_ij） ----------
xy = gdf[["cx", "cy"]].to_numpy()
D = np.sqrt(((xy[:, None, :] - xy[None, :, :]) ** 2).sum(-1))
assert np.allclose(np.diag(D), 0) and np.allclose(D, D.T)
print(f"[2] 距离矩阵对角为 0、对称  ✅\n{D.round(2)}")

# ---------- 3. SGS + 行归一化（论文 §2.1.3 式(1)） ----------
V = areas
with np.errstate(divide="ignore", invalid="ignore"):
    SGS = np.where(D == 0, 0.0, V[:, None] / np.where(D == 0, np.inf, D ** 2))
P = SGS / SGS.sum(axis=1, keepdims=True)
assert np.allclose(P.sum(axis=1), 1.0) and np.allclose(np.diag(P), 0.0)
print(f"[3] SGS 行归一化后每行和为 1、对角为 0  ✅")
print(f"    入流（列和）={P.sum(axis=0).round(3).tolist()}  ← 300 那栋最高")

# ---------- 4. 谱半径（G02 / G08 的收敛判据） ----------
rng = np.random.default_rng(0)
A = (rng.random((20, 20)) < 0.25).astype(float)
np.fill_diagonal(A, 0.0)
rho = float(np.max(np.abs(np.linalg.eigvals(A))))
print(f"[4] 20 节点邻接矩阵谱半径 {rho:.3f}  ✅（G08 用它判断迭代会不会发散）")

# ---------- 5. 连通分量与边数（论文 §3.1 图密度观察） ----------
G = nx.from_numpy_array(A)
print(f"[5] networkx 连通分量 {nx.number_connected_components(G)} 个、边 {G.number_of_edges()} 条  ✅")

# ---------- 6. Kaplan–Meier（G17 的折扣因子） ----------
T = np.array([2, 3, 3, 5, 6, 8, 8, 10, 12, 15], dtype=float)
E = np.array([1, 1, 0, 1, 1, 1, 0, 1, 1, 0])
kmf = KaplanMeierFitter().fit(T, event_observed=E)
print(f"[6] K-M 中位生存步数 {kmf.median_survival_time_}  ✅（生成轨迹的终止概率由它推）")

print("\n✅ 第一批依赖全部可用")
```

`D:\gpu_check.py` —— GPU 与矩阵规模实测（决定 G08／G16 敢不敢放大规模）：

```python
"""GPU 冒烟测试：确认 CUDA 可用，并实测论文量级（1014×1014）的矩阵乘。
跑法：conda activate geoirl && python D:\gpu_check.py
"""
import time
import torch

print(f"torch {torch.__version__} | built with CUDA {torch.version.cuda}")
print(f"可用 {torch.cuda.is_available()}")
assert torch.cuda.is_available(), "CUDA 不可用：检查是否装成了 CPU 版 torch"

print(f"设备 {torch.cuda.get_device_name(0)}")
print(f"计算能力 {torch.cuda.get_device_capability(0)}")
# 教训：mem_get_info() 返回 (空闲, 总量)，别按 (总量, 空闲) 读
free, total = torch.cuda.mem_get_info()
print(f"显存 mem_get_info() -> 空闲 {free / 1024**3:.2f} GB / 总 {total / 1024**3:.2f} GB")

N = 1014  # 论文学习子集的建筑数
a = torch.randn(N, N)
b = torch.randn(N, N)

t0 = time.perf_counter()
for _ in range(20):
    _ = a @ b
t_cpu = time.perf_counter() - t0
print(f"\nCPU  {t_cpu:.4f} s")

a, b = a.cuda(), b.cuda()
torch.cuda.synchronize()
t0 = time.perf_counter()
for _ in range(20):
    _ = a @ b
torch.cuda.synchronize()
t_gpu = time.perf_counter() - t0
print(f"CUDA {t_gpu:.4f} s   → 约 {t_cpu / t_gpu:.0f} 倍")

# MaxEnt IRL 每轮都要算 logsumexp，单独量一下
x = torch.randn(N, N, device="cuda")
torch.cuda.synchronize()
t0 = time.perf_counter()
for _ in range(200):
    torch.logsumexp(x, dim=1)
torch.cuda.synchronize()
print(f"200 轮 logsumexp({N},{N})  {time.perf_counter() - t0:.3f} s")
```
