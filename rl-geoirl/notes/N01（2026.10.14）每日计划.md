#### 参考代码

实际学习时手敲或保存为 `rl/N01/test_simple_gravity.py`。

```python
"""N01：独立执行官方 simple_gravity 函数节选，测试合成建筑输入。
下面simple_gravity是本地官方函数的原样节选；只避开整个transitions模块顶层的地图绘图依赖，计算未改。它测试该函数，不代表官方模块全量导入、数据处理或训练已跑通。
"""
import numpy as np
import pandas as pd
import torch as t


def simple_gravity (buildings : pd.DataFrame):
    beta = 2
    
    coor = t.tensor(buildings.loc[:, ["x", "y"]].to_numpy())
    vol = t.tensor(buildings.floorspace.to_numpy())
    distance = t.cdist(coor, coor, 2).pow(beta)
    distance[distance == 0] = 1e5
    scores = vol.div(distance)
    scores = scores.fill_diagonal_(0)
    transition = scores.div(t.sum(scores, 1).view(-1, 1)) 

    return transition


def main():
    buildings = pd.DataFrame({
        "x": [0.0, 1.0, 3.0],
        "y": [0.0, 0.0, 0.0],
        "floorspace": [10.0, 20.0, 30.0],
    })
    transition = simple_gravity(buildings)
    values = transition.detach().cpu().numpy()

    assert values.shape == (3, 3)
    assert np.isfinite(values).all()
    assert (values >= 0).all()
    np.testing.assert_allclose(np.diag(values), 0.0, atol=1e-8)
    np.testing.assert_allclose(values.sum(axis=1), 1.0, atol=1e-8)

    # 从建筑0出发：未归一化分数为20/1与30/9。
    manual_before = np.array([0.0, 6.0 / 7.0, 1.0 / 7.0])
    np.testing.assert_allclose(values[0], manual_before,
                               rtol=1e-6, atol=1e-8)

    # 只改变目的建筑2的面积，其他条件保持不变。
    changed = buildings.copy()
    changed.loc[2, "floorspace"] = 60.0
    values_after = simple_gravity(changed).detach().cpu().numpy()
    manual_after = np.array([0.0, 3.0 / 4.0, 1.0 / 4.0])
    np.testing.assert_allclose(values_after[0], manual_after,
                               rtol=1e-6, atol=1e-8)
    assert values_after[0, 2] > values[0, 2]

    print("合成输入：")
    print(buildings.to_string(index=False))
    print("原转移矩阵：")
    print(values)
    print("各行之和：", values.sum(axis=1))
    print("第一行修改前：", values[0])
    print("第一行修改后：", values_after[0])
    print("与手算的最大误差：",
          np.max(np.abs(values[0] - manual_before)),
          np.max(np.abs(values_after[0] - manual_after)))
    print("上述断言全部通过；这只是simple_gravity小验证。")


if __name__ == "__main__":
    main()
```

#### 代码与输出说明

- `cdist`计算建筑间距离，`pow(2)`对应距离平方。
- `vol.div(distance)`沿目的列使用建筑面积，不能把旧笔记的来源面积公式直接套过来。
- 归一化前是吸引力分数，行归一化后是本函数的移动权重。
- 增大建筑2面积后，从建筑0到建筑2的相对权重上升；这仍是构图计算，不是IRL已经学得的策略。
- `get_transition`后面还做阈值和`row_sum+1`归一化，不应要求它每行也等于1。
