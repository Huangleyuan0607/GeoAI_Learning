1、**`model.py`**（CNN 分类模型定义）：

```python
import torch
import torch.nn as nn

class ConvClassifier(nn.Module):
    """用于图像分类的简单 CNN 网络"""
    def __init__(self, num_classes=10):
        super(ConvClassifier, self).__init__()
        self.features = nn.Sequential(
            nn.Conv2d(in_channels=1, out_channels=16, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2),  # (B, 16, 14, 14)
            nn.Conv2d(in_channels=16, out_channels=32, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2)   # (B, 32, 7, 7)
        )
        self.classifier = nn.Sequential(
            nn.Linear(32 * 7 * 7, 128),
            nn.ReLU(),
            nn.Linear(128, num_classes)
        )

    def forward(self, x):
        x = self.features(x)
        x = x.view(x.size(0), -1)
        x = self.classifier(x)
        return x
```

2、**`train.py`**（模拟训练与验证流程）：

```python
import torch
import torch.nn as nn
import torch.optim as optim
from model import ConvClassifier

def train_demo():
    # 模拟构建 (Batch_size=8, Channel=1, Height=28, Width=28) 的图像数据与标签
    dummy_inputs = torch.randn(32, 1, 28, 28)
    dummy_labels = torch.randint(0, 10, (32,))

    model = ConvClassifier(num_classes=10)
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(model.parameters(), lr=0.001)

    model.train()
    for epoch in range(1, 4):
        optimizer.zero_grad()
        outputs = model(dummy_inputs)
        loss = criterion(outputs, dummy_labels)
        loss.backward()
        optimizer.step()
        
        print(f"Epoch [{epoch}/3], Loss: {loss.item():.4f}")

if __name__ == "__main__":
    train_demo()
```
