### 第二阶段：理论学习（约 0.5 小时）

深入理解 PyTorch 神经网络架构与训练机制：  

- **`nn.Module` 核心结构**： 所有的自定义神经网络必须继承 `torch.nn.Module`，并在 `__init__` 中调用 `super().__init__()`。定义在 `__init__` 中的层（如 `nn.Linear`）其参数（`weight` 与 `bias`）会被自动注册到模型参数列表中。  

- **前向传播与反向传播**： 直接调用 `model(x)` 会触发 `nn.Module` 的 `__call__` 方法，内部会自动执行 `forward(x)`。梯度通过损失函数的 `loss.backward()` 计算，优化器 `optimizer.step()` 完成权重的调整。  

- **训练循环标准范式**：  

  ```
  optimizer.zero_grad()  # 1. 梯度清空
  outputs = model(x)     # 2. 前向传播
  loss = criterion(outputs, y) # 3. 计算损失
  loss.backward()        # 4. 反向传播
  optimizer.step()       # 5. 参数更新
  ```

#### 实验任务

创建 `mlp_classifier.py` 文件，编写并运行以下代码，实现多层感知机（MLP）三分类网络的完整搭建、合成数据生成、训练循环以及 Loss/Accuracy 曲线可视化：  

```python
"""
程序名称：mlp_classifier.py
作者：黄乐源
日期：2026-07-22
描述：PyTorch 多层感知机 (MLP) 分类网络完整搭建、数据集加载、模型训练与准确率测试实战
"""

import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import TensorDataset, DataLoader
import matplotlib.pyplot as plt

print("================= 任务 1：构建分类数据集 =================")

# 随机生成 3 簇 2 维分类数据（模拟 3 分类任务）
torch.manual_seed(42)

n_samples = 300
class_0 = torch.randn(n_samples, 2) + torch.tensor([-2.0, -2.0])
class_1 = torch.randn(n_samples, 2) + torch.tensor([2.0, 2.0])
class_2 = torch.randn(n_samples, 2) + torch.tensor([-2.0, 2.0])

X = torch.cat([class_0, class_1, class_2], dim=0)
y = torch.cat([torch.zeros(n_samples), torch.ones(n_samples), torch.full((n_samples,), 2)], dim=0).long()

# 划分训练集与测试集 (8:2)
indices = torch.randperm(X.size(0))
X, y = X[indices], y[indices]

train_size = int(0.8 * len(X))
X_train, X_test = X[:train_size], X[train_size:]
y_train, y_test = y[:train_size], y[train_size:]

# 构造 DataLoader
train_dataset = TensorDataset(X_train, y_train)
train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)

print(f"训练集样本数: {len(X_train)}, 测试集样本数: {len(X_test)}")

print("\n================= 任务 2：搭建 MLP 分类网络 =================")

class MLPClassifier(nn.Module):
    def __init__(self, input_dim, hidden_dim, num_classes):
        super(MLPClassifier, self).__init__()
        self.fc1 = nn.Linear(input_dim, hidden_dim)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(hidden_dim, hidden_dim)
        self.fc3 = nn.Linear(hidden_dim, num_classes)

        # 参数初始化
        nn.init.kaiming_normal_(self.fc1.weight, nonlinearity='relu')
        nn.init.kaiming_normal_(self.fc2.weight, nonlinearity='relu')
        nn.init.xavier_uniform_(self.fc3.weight)

    def forward(self, x):
        x = self.relu(self.fc1(x))
        x = self.relu(self.fc2(x))
        out = self.fc3(x)  # 注意：CrossEntropyLoss 内部集成 Softmax
        return out

model = MLPClassifier(input_dim=2, hidden_dim=16, num_classes=3)
print("网络结构:\n", model)

print("\n================= 任务 3：模型训练与评估 =================")

criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=0.01)

epochs = 50
loss_history = []
acc_history = []

for epoch in range(1, epochs + 1):
    model.train()
    running_loss = 0.0
    
    for batch_x, batch_y in train_loader:
        # 1. 清空梯度
        optimizer.zero_grad()
        # 2. 前向传播
        outputs = model(batch_x)
        # 3. 计算损失
        loss = criterion(outputs, batch_y)
        # 4. 反向传播
        loss.backward()
        # 5. 参数更新
        optimizer.step()
        
        running_loss += loss.item() * batch_x.size(0)
        
    epoch_loss = running_loss / len(X_train)
    loss_history.append(epoch_loss)

    # 每个 Epoch 在测试集上评估准确率
    model.eval()
    with torch.no_grad():
        test_outputs = model(X_test)
        preds = torch.argmax(test_outputs, dim=1)
        correct = (preds == y_test).sum().item()
        accuracy = correct / len(y_test)
        acc_history.append(accuracy)

    if epoch % 10 == 0 or epoch == 1:
        print(f"Epoch [{epoch:02d}/{epochs}] - Loss: {epoch_loss:.4f} - Test Acc: {accuracy * 100:.2f}%")

print("\n================= 任务 4：训练曲线可视化 =================")

plt.figure(figsize=(12, 4))

plt.subplot(1, 2, 1)
plt.plot(range(1, epochs + 1), loss_history, color='blue', linewidth=2)
plt.title("Training Loss Curve")
plt.xlabel("Epoch")
plt.ylabel("CrossEntropy Loss")
plt.grid(True)

plt.subplot(1, 2, 2)
plt.plot(range(1, epochs + 1), acc_history, color='green', linewidth=2)
plt.title("Test Accuracy Curve")
plt.xlabel("Epoch")
plt.ylabel("Accuracy")
plt.grid(True)

plt.tight_layout()
plt.show()

print("\n🏁 Day25 MLP 分类网络训练实战成功完成！")
```
