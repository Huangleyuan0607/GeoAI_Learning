#### 实验任务

创建 `Day19_tensor_numpy.py` 文件，手工编写并运行以下代码，实现 Tensor 进阶创建、类型转换及与 NumPy 互转：

```python
"""
程序名称：Day19_tensor_numpy.py
作者：黄乐源
日期：2026-07-16
描述：PyTorch Tensor 常用创建进阶、数据类型转换与 NumPy 双向转换实战（结合 GeoAI 遥感/GIS 背景）
"""

import numpy as np
import torch

print("================= 任务 1：指定值、线性与随机 Tensor 创建 =================")

# 1. 创建全 0 掩码矩阵 (如遥感图像背景 Mask)
mask_zeros = torch.zeros((3, 3), dtype=torch.uint8)
print(f"3x3 全 0 掩码张量:\n{mask_zeros}")

# 2. 创建特定初始值的张量 (如初始化偏置项或特定高程填充)
padding_tensor = torch.full((2, 4), fill_value=-9999.0)
print(f"\n指定填充值 (-9999.0) 张量:\n{padding_tensor}")

# 3. 线性等差序列 (如波段波长采样点或时间序列索引)
wavelengths = torch.linspace(start=400, end=900, steps=6)
print(f"\n400nm-900nm 线性采样波段序列:\n{wavelengths}")

# 4. 正态分布随机数 (模拟遥感影像传感器高斯噪声或权重初始化)
gaussian_noise = torch.randn((2, 3))
print(f"\n标准正态分布随机噪声张量:\n{gaussian_noise}")

print("\n================= 任务 2：Tensor 数据类型（Dtype）转换 =================")

# 模拟原始 8 位无符号整型遥感单波段数据 (0-255)
raw_raster = torch.tensor([[50, 120, 200], [80, 150, 220]], dtype=torch.uint8)
print(f"原始 uint8 栅格张量类型: {raw_raster.dtype}")

# 归一化输入前需转换为 float32 浮点型
float_raster = raw_raster.to(torch.float32) / 255.0
print(f"转换并归一化后的 float32 栅格张量:\n{float_raster}")
print(f"新张量类型: {float_raster.dtype}")

print("\n================= 任务 3：Tensor 与 NumPy 数组互转及内存共享校验 =================")

# 1. NumPy Array -> PyTorch Tensor
np_array = np.array([101.5, 102.8, 104.3], dtype=np.float32)
tensor_from_np = torch.from_numpy(np_array)
print(f"从 NumPy 转换得到的 Tensor:\n{tensor_from_np}")

# 2. PyTorch Tensor -> NumPy Array
tensor_data = torch.ones((2, 2), dtype=torch.float32)
np_from_tensor = tensor_data.numpy()
print(f"从 Tensor 转换得到的 NumPy 数组:\n{np_from_tensor}")

# 3. 验证内存共享特性
np_array[0] = 999.9
print(f"\n修改 NumPy 原数组后，Tensor 同步更新结果:\n{tensor_from_np}")

print("\n🏁 Day19 GeoAI Tensor 进阶创建与 NumPy 互转实战完成！")
```
