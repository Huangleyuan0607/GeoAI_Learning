  - **BatchNorm 的作用**：

    - 加速训练
    - 稳定梯度
    - 提升泛化能力

  - **训练阶段**：用当前 batch 的统计量做标准化

    ```
    mean
    variance
    ```

  - **推理阶段**：用训练过程中累计的滑动平均

    ```
    running mean
    running variance
    ```

  - **两个阶段的对比**：

    | 阶段 | 均值 / 方差来源 | 特点 |
    | ---- | -------------- | ---- |
    | 训练 | 当前 batch 的 `mean` / `variance` | 每个 batch 重新统计，同时更新 `running mean` / `running variance` |
    | 推理 | `running mean` / `running variance` | 不再依赖 batch，单张图片也能前向 |

#### 参考代码

创建 `BatchNorm.py`，从零实现 BN 层，并对比"有 BN / 无 BN"两个模型在相同数据与超参数下的训练表现：

```python
# 导包
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, Subset
from torchvision.datasets import FashionMNIST
from torchvision.transforms import ToTensor


# 1.从零实现批量归一化层（对应李沐第92讲）
class MyBatchNorm(nn.Module):
    """批量归一化：训练阶段用当前batch的统计量，推理阶段用累计的滑动平均统计量"""
    # 参1：特征数（卷积层传通道数，全连接层传输出特征数）；参2：数据维度（4表示(N, C, H, W)，2表示(N, C)）
    def __init__(self, num_features, num_dims = 4, eps = 1e-5, momentum = 0.1):
        super(MyBatchNorm, self).__init__()
        # 1.定义可学习的缩放参数gamma与平移参数beta，对标准化时被抹掉的信息做补偿
        if num_dims == 2:       # 全连接层输出：(N, C)
            shape = (1, num_features)
        else:                   # 卷积层输出：(N, C, H, W)
            shape = (1, num_features, 1, 1)
        self.gamma = nn.Parameter(torch.ones(shape))
        self.beta = nn.Parameter(torch.zeros(shape))
        # 2.定义训练过程中累计的滑动平均统计量（不参与梯度更新，但要跟着模型搬到GPU上）
        self.register_buffer('moving_mean', torch.zeros(shape))
        self.register_buffer('moving_var', torch.ones(shape))
        self.num_dims = num_dims
        self.eps = eps
        self.momentum = momentum

    def forward(self, X):
        # 1.推理阶段：直接用训练时累计的running mean / running var，不再依赖当前batch
        if not self.training:
            X_hat = (X - self.moving_mean) / torch.sqrt(self.moving_var + self.eps)
        # 2.训练阶段：用当前batch的统计量做标准化
        else:
            # 2.1按通道求均值与方差：4维在(N, H, W)上求，2维在N上求
            if self.num_dims == 2:
                mean = X.mean(dim = 0)
                var = ((X - mean) ** 2).mean(dim = 0)
            else:
                mean = X.mean(dim = (0, 2, 3), keepdim = True)
                var = ((X - mean) ** 2).mean(dim = (0, 2, 3), keepdim = True)
            # 2.2标准化
            X_hat = (X - mean) / torch.sqrt(var + self.eps)
            # 2.3累计滑动平均统计量（推理阶段要用），detach()避免统计量被计入计算图
            self.moving_mean = self.momentum * self.moving_mean + (1.0 - self.momentum) * mean.detach()
            self.moving_var = self.momentum * self.moving_var + (1.0 - self.momentum) * var.detach()
        # 3.缩放 + 平移
        return self.gamma * X_hat + self.beta


# 2.定义小型CNN（LeNet结构），use_bn控制是否在卷积/全连接层后接批量归一化
class SmallCNN(nn.Module):
    def __init__(self, use_bn = False, num_classes = 10):
        super(SmallCNN, self).__init__()
        self.use_bn = use_bn
        # 1.卷积部分：Conv -> (BN) -> ReLU -> MaxPool
        self.conv1 = nn.Conv2d(1, 6, kernel_size = 5, padding = 2)
        self.conv2 = nn.Conv2d(6, 16, kernel_size = 5)
        # 2.全连接部分
        self.fc1 = nn.Linear(16 * 5 * 5, 120)
        self.fc2 = nn.Linear(120, 84)
        self.fc3 = nn.Linear(84, num_classes)
        # 3.BN层：只有use_bn为True时才创建（用从零实现的MyBatchNorm，也可以换成nn.BatchNorm2d / nn.BatchNorm1d）
        if use_bn:
            self.bn1 = MyBatchNorm(6)               # 卷积层输出是4维(N, C, H, W)
            self.bn2 = MyBatchNorm(16)
            self.bn3 = MyBatchNorm(120, num_dims = 2)   # 全连接层输出是2维(N, C)
            self.bn4 = MyBatchNorm(84, num_dims = 2)

    def forward(self, x):
        # 1.卷积块1：Conv(1->6) + MaxPool，28x28 -> 14x14
        x = self.conv1(x)
        if self.use_bn:
            x = self.bn1(x)
        x = torch.relu(x)
        x = torch.max_pool2d(x, kernel_size = 2, stride = 2)
        # 2.卷积块2：Conv(6->16) + MaxPool，14x14 -> 5x5
        x = self.conv2(x)
        if self.use_bn:
            x = self.bn2(x)
        x = torch.relu(x)
        x = torch.max_pool2d(x, kernel_size = 2, stride = 2)
        # 3.展平后接全连接层：Linear -> (BN) -> ReLU
        x = x.reshape(x.shape[0], -1)       # (N, 16, 5, 5) -> (N, 400)
        x = self.fc1(x)
        if self.use_bn:
            x = self.bn3(x)
        x = torch.relu(x)
        x = self.fc2(x)
        if self.use_bn:
            x = self.bn4(x)
        x = torch.relu(x)
        return self.fc3(x)


# 3.定义数据加载函数（复用项目里已有的FashionMNIST，取子集保证运行快）
def get_data_loaders(batch_size = 64):
    transform = ToTensor()
    # 数据已在notebooks/data目录下，download = False表示不再联网下载
    train_dataset = FashionMNIST(root = './notebooks/data', train = True, transform = transform, download = False)
    test_dataset = FashionMNIST(root = './notebooks/data', train = False, transform = transform, download = False)
    # 只取一部分数据：6000张训练 + 1000张测试
    train_dataset = Subset(train_dataset, range(6000))
    test_dataset = Subset(test_dataset, range(1000))
    train_loader = DataLoader(train_dataset, batch_size = batch_size, shuffle = True)
    test_loader = DataLoader(test_dataset, batch_size = batch_size, shuffle = False)
    return train_loader, test_loader


# 4.定义训练一个epoch与测试的函数
def train_one_epoch(model, train_loader, criterion, optimizer, device):
    model.train()
    total_loss, total_num = 0.0, 0
    for X, y in train_loader:
        X, y = X.to(device), y.to(device)
        optimizer.zero_grad()
        y_hat = model(X)
        loss = criterion(y_hat, y)
        loss.backward()
        optimizer.step()
        total_loss += loss.item() * X.shape[0]
        total_num += X.shape[0]
    return total_loss / total_num


def evaluate(model, test_loader, device):
    model.eval()        # 推理模式：BN层改用running统计量
    correct, total = 0, 0
    with torch.no_grad():
        for X, y in test_loader:
            X, y = X.to(device), y.to(device)
            pred = model(X).argmax(dim = 1)
            correct += (pred == y).sum().item()
            total += y.shape[0]
    return correct / total


# 5.定义主函数
def main():
    # 1.设置随机种子、设备与超参数
    torch.manual_seed(42)
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    lr, epochs = 0.1, 5
    train_loader, test_loader = get_data_loaders(batch_size = 64)
    criterion = nn.CrossEntropyLoss()

    # 2.验证从零实现的BN层：标准化后每个通道均值≈0、方差≈1
    print("=" * 60)
    print("1.从零实现BN层的效果验证")
    print("=" * 60)
    bn = MyBatchNorm(3)
    X = torch.randn(4, 3, 8, 8) * 5 + 2     # 人为放大并平移，模拟"分布跑偏"的中间层输出
    Y = bn(X)
    print(f"标准化前 各通道均值：{[round(v, 4) for v in X.mean(dim = (0, 2, 3)).tolist()]}")
    print(f"标准化前 各通道方差：{[round(v, 4) for v in X.var(dim = (0, 2, 3), unbiased = False).tolist()]}")
    print(f"标准化后 各通道均值：{[round(v, 4) for v in Y.mean(dim = (0, 2, 3)).tolist()]}")
    print(f"标准化后 各通道方差：{[round(v, 4) for v in Y.var(dim = (0, 2, 3), unbiased = False).tolist()]}")

    # 3.对比训练阶段与推理阶段的统计量来源
    print()
    print("=" * 60)
    print("2.训练阶段与推理阶段的统计量来源")
    print("=" * 60)
    bn_demo = MyBatchNorm(3)
    print(f"训练前 running mean：{[round(v, 4) for v in bn_demo.moving_mean.reshape(-1).tolist()]}")
    print(f"训练前 running var： {[round(v, 4) for v in bn_demo.moving_var.reshape(-1).tolist()]}")
    bn_demo.train()         # 训练模式：用当前batch统计量做标准化，并更新running统计量
    for _ in range(5):      # 模拟训练5个batch
        bn_demo(torch.randn(4, 3, 8, 8) * 5 + 2)
    print(f"训练5个batch后 running mean：{[round(v, 4) for v in bn_demo.moving_mean.reshape(-1).tolist()]}")
    print(f"训练5个batch后 running var： {[round(v, 4) for v in bn_demo.moving_var.reshape(-1).tolist()]}")
    bn_demo.eval()          # 推理模		式：不再看当前batch，直接用running统计量
    X_test = torch.randn(2, 3, 8, 8) * 5 + 2
    print(f"推理阶段：输入均值 {X_test.mean().item():.4f} -> 用running统计量标准化后输出均值 {bn_demo(X_test).mean().item():.4f}")

    # 4.参数量对比（BN层每个通道多2个可学习参数：gamma与beta）
    print()
    print("=" * 60)
    print("3.有BN与无BN的模型参数量对比")
    print("=" * 60)
    for use_bn in (False, True):
        net = SmallCNN(use_bn = use_bn)
        total = sum(p.numel() for p in net.parameters())
        print(f"use_bn = {str(use_bn):<6}总参数量：{total:,}")

    # 5.训练对比：同样的数据、同样的学习率，两个模型只差一组BN层
    print()
    print("=" * 60)
    print(f"4.训练对比（FashionMNIST 6000张训练 / 1000张测试，lr = {lr}，epochs = {epochs}）")
    print("=" * 60)
    results = {}
    for use_bn in (False, True):
        name = "有BN" if use_bn else "无BN"
        torch.manual_seed(42)       # 两个模型用同一个初始化种子，尽量保证对比公平
        model = SmallCNN(use_bn = use_bn).to(device)
        optimizer = optim.SGD(model.parameters(), lr = lr)
        history = []
        print(f"---------- {name} ----------")
        for epoch in range(1, epochs + 1):
            train_loss = train_one_epoch(model, train_loader, criterion, optimizer, device)
            test_acc = evaluate(model, test_loader, device)
            history.append((train_loss, test_acc))
            print(f"epoch {epoch}  train_loss = {train_loss:.4f}  test_acc = {test_acc:.4f}")
        results[name] = history

    # 6.汇总对比
    print()
    print("=" * 60)
    print("5.结果汇总")
    print("=" * 60)
    print(f"{'epoch':<8}{'无BN loss':>12}{'无BN acc':>12}{'有BN loss':>12}{'有BN acc':>12}")
    for i in range(epochs):
        no_bn_loss, no_bn_acc = results['无BN'][i]
        bn_loss, bn_acc = results['有BN'][i]
        print(f"{i + 1:<8}{no_bn_loss:>12.4f}{no_bn_acc:>12.4f}{bn_loss:>12.4f}{bn_acc:>12.4f}")

    print()
    print("结论：同样的数据与超参数下，加了BN的网络收敛更快、曲线更平稳，说明BN确实能加速训练、稳定梯度。")


# 6.测试
if __name__ == '__main__':
    main()
```

  - 固定小批量里面的均值和方差
    $$
    \mu_B = \frac{1}{|B|} \sum_{i \in B} x_i \quad \text{and} \quad \sigma_B^2 = \frac{1}{|B|} \sum_{i \in B} (x_i - \mu_B)^2 + \epsilon
    $$
    然后再做额外的调整（可学习的参数）
    $$
    x_{i+1} = \gamma \frac{x_i - \mu_B}{\sigma_B} + \beta
    $$

  - 实则就是黑马里提到的**先对数据做标准化（归一化），再对数据进行缩放和平移**

- **批量归一化层**

  - 可学系的参数为γ和β
  - 作用在
    - 全连接层和卷积层输出上，激活函数前
    - 全连接层和卷积层输入上
  - 对**全连接层**，作用在**特征**维
  - 对于**卷积层**，作用在**通道**维

- 批量归一化的**作用**

  - 最初是想用它来减少内部协变量转移

  - 后续有论文指出它可能就是通过在每个小批量里加入噪音（**随机的`μ_b`和`σ_b`**）来控制模型复杂度
    $$
    x_{i+1} = \gamma \frac{x_i - \hat{\mu}_B}{\hat{\sigma}_B} + \beta
    $$
  
- 因此**没必要跟Dropout法（丢弃法）混合使用**
  
- 总结

  - 批量归一化固定小批量中的均值和方差（），然后学习出适合的**偏移和缩放**
  - 可以**加速收敛速度**，但一般**不改变模型精度**
